## Solution

```java
class Solution {
  public List<List<Integer>> permuteUnique(int[] nums) {
    Arrays.sort(nums);
    boolean[] visited = new boolean[nums.length];
    List<List<Integer>> ans = new ArrayList<>();
    List<Integer> ds = new ArrayList<>();
    rec(visited, nums, ds, ans);
    return ans;
  }

  private void rec(boolean[] visited, int[] nums, List<Integer> ds, List<List<Integer>> ans) {
    //basecondition
    if (ds.size() == nums.length) {
      ans.add(new ArrayList<>(ds));
      return;
    }

    for (int i = 0; i < nums.length; i++) {
      if (visited[i])
        continue;
      if (i > 0 && nums[i] == nums[i - 1] && !visited[i - 1])
        continue;
      
      ds.add(nums[i]);
      visited[i] = true;
      rec(visited, nums, ds, ans);
      visited[i] = false;
      ds.remove(ds.size() - 1);

    }
  }

}
```

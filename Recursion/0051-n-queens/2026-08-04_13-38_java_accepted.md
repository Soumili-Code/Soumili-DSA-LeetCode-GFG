class Solution {
    public List<List<String>> solveNQueens(int n) {
        List<List<String>> ans = new ArrayList<>();
        char[][] board = new char[n][n];

        for (int i = 0; i < n; i++) {
          Arrays.fill(board[i], '.');
        }
        rec(0,n,board,ans);
        return ans;
    }
    public void rec(int i,int n,char[][] board, List<List<String>> ans){
      if(i==n){
        List<String> path = constructBoard(n,board);
        ans.add(path);
        return;
      } // base condition
      for(int col=0;col<n;col++){
        if(isSafePlace(i,col,n,board)){
          board[i][col]='Q';
          rec(i+1,n,board,ans);
          board[i][col]='.';
        }
      }
    }
    public boolean isSafePlace(int row,int col,int n,char[][] board){
        for(int i=row-1;i>=0;i--){ //top rows checking 
          if(board[i][col]=='Q'){
            return false;
          }
        }
        int i=row-1,j=col-1;  //top-left checking 
        while(i>=0 && j>=0){
           if(board[i][j]=='Q'){
            return false;
          }
          i--;
          j--;
        }
        i=row-1;
        j=col+1;
        while(i>=0 && j<n){ //top-right 
          if(board[i][j]=='Q'){
            return false;
          }
          i--;
          j++;
        }
        return true;

    }
    public List<String> constructBoard(int n, char[][] board){
      List<String> path = new ArrayList<>();
      for(int i=0;i<n;i++){
        String temp = "";
        for(int j=0;j<n;j++){
          temp+=board[i][j]; 
        }
        path.add(temp);
      }
      return path;
    } 

}
